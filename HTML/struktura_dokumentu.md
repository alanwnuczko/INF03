### `<html>`
```html
<html lang="pl">
  <!-- Cały dokument -->
</html>
```
 
### `<head>`
```html
<head> <!-- Sekcja konfiguracyjna Strony -->
  <meta charset="UTF-8">
  <title>Tytuł strony</title>
</head>
```
 
### `<body>`
```html
<body>
  <!-- Widoczna treść strony -->
</body>
```
 
### `<title>`
```html
<title>Strona internetowa</title> <!-- Tytuł strony widoczny na karcie przeglądarki -->
```

### Przykład

```html
<!DOCTYPE html>
<html lang="pl">
<head>
    <meta charset="UTF-8"> <!-- Kodowanie polskich znaków -->
    <link rel="stylesheet" href="style.css"> <!-- Podłączenie arkusza stylów -->
    <title>Document</title> <!-- Tytuł strony widoczny na karcie przeglądarki -->
</head>
<body>
    

    <!-- Widoczna treść strony -->


    <script src="skrypt.js"></script> <!-- Podłączenie pliku z skryptem JS -->
</body>
</html>
```
